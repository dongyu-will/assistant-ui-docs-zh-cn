### useAssistantFrameHost

```typescript

  const iframeRef = useRef<HTMLIFrameElement>(null);

  useAssistantFrameHost({
    iframeRef,
    targetOrigin: "https://trusted-domain.com",

  return <iframe ref={iframeRef} src="..." />;
}
```

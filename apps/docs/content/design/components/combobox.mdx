两种风格都封装了 Base UI 的 Combobox；Radix 不提供 combobox 原语。输入内容会根据传递给根组件的 `items` 缩小列表范围；如果没有匹配项，弹出层会显示其空状态。

## 安装

<KitInstall name="combobox" />

## 用法

```tsx
import {
  Combobox,
  ComboboxContent,
  ComboboxEmpty,
  ComboboxInput,
  ComboboxItem,
  ComboboxList,
} from "@/components/ui/combobox";

const runtimes = ["AI SDK", "LangGraph", "LangChain", "Mastra"];

export function RuntimePicker() {
  return (
    <Combobox items={runtimes}>
      <ComboboxInput placeholder="Search runtimes…" />
      <ComboboxContent>
        <ComboboxEmpty>No runtimes found.</ComboboxEmpty>
        <ComboboxList>
          {(runtime: string) => (
            <ComboboxItem key={runtime} value={runtime}>
              {runtime}
            </ComboboxItem>
          )}
        </ComboboxList>
      </ComboboxContent>
    </Combobox>
  );
}
```
